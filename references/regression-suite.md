### 一键复跑（28 题；以最新生成器 meta.slots 为准）

在回归语料根目录（`exam-helper/data-vis/skill-audit-2026-11-26/`，外部语料，不在本仓库内）执行；把 `CHK` 指向本仓库的 `scripts/check_output.mjs` 即可。脚本会读每题 `check.json` 的 `type/blanks/answerTerms`，再用 `preflight.jsonl` 的 `answer_slots` 覆盖：

```bash
python3 - <<'PY'
import json, os, re, subprocess
CHK = "/path/to/shenti-coach/scripts/check_output.mjs"  # 改成你的本仓库路径
pre = {json.loads(l)["id"]: json.loads(l) for l in open("preflight.jsonl")}
for d in ["regression", "semantic-batch2", "semantic-batch3", "semantic-batch4", "semantic-batch5"]:
    for x in sorted(os.listdir(d)):
        p = f"{d}/{x}"
        if not (os.path.isdir(p) and os.path.exists(f"{p}/check.json")):
            continue
        m = re.match(r"(math|physics|chem|bio)-(bj|ln)(\\d{4})-q(.+)", x)
        subject = {"math": "数学", "physics": "物理", "chem": "化学", "bio": "生物"}[m.group(1)]
        region = {"bj": "北京", "ln": "辽宁"}[m.group(2)]
        row = pre[f"{subject}-{region}-{m.group(3)}-q{m.group(4)}"]
        old = json.load(open(f"{p}/check.json"))
        meta = json.load(open(f"{p}/meta.json")) if os.path.exists(f"{p}/meta.json") else {}
        # preflight 对“一个 Q 内嵌 ①②③”会少计，生成器 meta.slots 才是该稿的实际 B 数。
        slots = max(int(meta.get("slots") or 0), int(row.get("answer_slots") or 0))
        cmd = ["node", CHK, f"{p}/output.md", "--type", meta.get("checkType") or old.get("type") or "solution"]
        blanks = meta.get("blanks", old.get("blanks"))
        if blanks is not None: cmd += ["--blanks", str(blanks)]
        if slots > 0: cmd += ["--answer-slots", str(slots)]
        qc = meta.get("qCount") or old.get("qCount")
        if qc: cmd += ["--q-count", str(qc)]
        if slots >= 2: cmd += ["--require-blank-map"]
        for a in old.get("answerTerms") or []: cmd += ["--answer", a]
        if old.get("answerTerms"): cmd += ["--strict-answer"]
        cmd += ["--require-media", "--json"]
        j = json.loads(subprocess.run(cmd, capture_output=True, text=True).stdout)
        json.dump(j, open(f"{p}/check.json", "w"), ensure_ascii=False, indent=2)
        print(p, "PASS" if j.get("pass") else "FAIL", len(j.get("fail", [])), len(j.get("warn", [])))
PY
```
